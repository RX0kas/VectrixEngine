55 inline bool loadOBJ(const std::string& filepath,std::vector<Vertex>& outVertices,std::vector<uint32_t>& outIndices) {
56 tinyobj::attrib_t attrib;
57 std::vector<tinyobj::shape_t> shapes;
58 std::vector<tinyobj::material_t> materials;
59
60 std::string err;
61
62 // Opened here rather than by tinyobj, which reads the name in the system code page on Windows (as its
63 // filename overload does, the .mtl files are looked for from the working directory)
64 std::ifstream stream(fromUtf8(filepath));
65 if (!stream) {
66 VC_CORE_ERROR_NO_EXIT("Failed to load OBJ {}: can't open the file",filepath);
67 return false;
68 }
69 tinyobj::MaterialFileReader materialReader("");
70 const bool ret = tinyobj::LoadObj(&attrib,&shapes,&materials,&err,&stream,&materialReader);
71
72
73 // This tinyobjloader version reports warnings (e.g. a missing .mtl file) through err too,
74 // so only the return value tells whether the load failed
75 if (!ret) {
76 VC_CORE_ERROR_NO_EXIT("Failed to load OBJ {}: {}",filepath,err);
77 return false;
78 }
79 if (!err.empty()) {
80 VC_CORE_WARN("OBJ {}: {}",filepath,err);
81 }
82
83 std::unordered_map<Vertex, uint32_t> uniqueVertices{};
84
85 for (const auto& shape : shapes) {
86 for (const auto& index : shape.mesh.indices) {
87 Vertex vertex{};
88
89 // POSITION
90 vertex.position = {
91 attrib.vertices[3 * index.vertex_index + 0],
92 attrib.vertices[3 * index.vertex_index + 1],
93 attrib.vertices[3 * index.vertex_index + 2]
94 };
95
96 // NORMAL
97 if (index.normal_index >= 0) {
98 vertex.normal = {
99 attrib.normals[3 * index.normal_index + 0],
100 attrib.normals[3 * index.normal_index + 1],
101 attrib.normals[3 * index.normal_index + 2]
102 };
103 } else {
104 vertex.normal = glm::vec3(0.0f);
105 }
106
107 // TEXCOORD
108 if (index.texcoord_index >= 0) {
109 vertex.uv = {
110 attrib.texcoords[2 * index.texcoord_index + 0],
111 1.0f - attrib.texcoords[2 * index.texcoord_index + 1]
112 };
113 } else {
114 vertex.uv = glm::vec2(0.0f);
115 }
116
117 // DEDUPLICATION
118 if (uniqueVertices.count(vertex) == 0) {
119 uniqueVertices[vertex] = static_cast<uint32_t>(outVertices.size());
120 outVertices.push_back(vertex);
121 }
122
123 outIndices.push_back(uniqueVertices[vertex]);
124 }
125 }
126
127 return true;
128 }
129}
