71 void Window::init(const WindowAttributes& attributes) {
72 VC_PROFILER_FUNCTION();
73 VC_CORE_INFO("Creating window {0} ({1}, {2})", Application::getAppInfo().getAppName(), attributes.width, attributes.height);
74
75 m_data.title = Application::getAppInfo().getAppName();
76 m_data.visible = false;

79#else
80 m_data.displayServer = detectLinuxDisplayServer();
81#endif

85 glfwWindowHint(GLFW_VISIBLE,GLFW_FALSE);
86 m_window = glfwCreateWindow(static_cast<int>(attributes.width), static_cast<int>(attributes.height), m_data.title.c_str(), nullptr, nullptr);
87 if (m_window == nullptr) {
88 VC_CORE_CRITICAL("Failed to create the GLFW window");
89 return;
90 }
91 glfwSetWindowUserPointer(m_window, &m_data);
92
93 // width/height track the framebuffer in pixels, which the swap chain has to match. On a scaled
94 // display (HiDPI, Wayland scale factor) it differs from the window size in screen coordinates.
95 int framebufferWidth = 0, framebufferHeight = 0;
96 glfwGetFramebufferSize(m_window, &framebufferWidth, &framebufferHeight);
97 m_data.width = static_cast<unsigned int>(framebufferWidth);
98 m_data.height = static_cast<unsigned int>(framebufferHeight);
99
100 // Set some callbacks
101 glfwSetFramebufferSizeCallback(m_window, framebufferResizeCallback);
102
103 // Without it GLFW never reports Caps Lock and Num Lock in the modifiers
104 glfwSetInputMode(m_window, GLFW_LOCK_KEY_MODS, GLFW_TRUE);
105
106 // Set before ImGui installs its own callbacks, which forward to these
107 glfwSetWindowSizeCallback(m_window, [](GLFWwindow* window, int width, int height) {
108 postEvent<WindowResizeEvent>(window, static_cast<unsigned int>(width), static_cast<unsigned int>(height));
109 });

112 postEvent<WindowCloseEvent>(window);
113 });

116 postEvent<WindowContentScaleEvent>(window, xScale, yScale);
117 });

136 glfwSetKeyCallback(m_window, [](GLFWwindow* window, int key, int scancode, int action, int mods) {
137 switch (action) {
138 case GLFW_PRESS: postEvent<KeyPressedEvent>(window, key, false, toKeyMods(mods), scancode); break;
139 case GLFW_REPEAT: postEvent<KeyPressedEvent>(window, key, true, toKeyMods(mods), scancode); break;
140 case GLFW_RELEASE: postEvent<KeyReleasedEvent>(window, key, toKeyMods(mods), scancode); break;
141 default: VC_CORE_ERROR_NO_EXIT("Unknown key action: {}", action);
142 }
143 });

145 glfwSetCharCallback(m_window, [](GLFWwindow* window, unsigned int codepoint) {

147 });

149 glfwSetMouseButtonCallback(m_window, [](GLFWwindow* window, int button, int action, int mods) {
150 switch (action) {
151 case GLFW_PRESS: postEvent<MouseButtonPressedEvent>(window, button, toKeyMods(mods)); break;
152 case GLFW_RELEASE: postEvent<MouseButtonReleasedEvent>(window, button, toKeyMods(mods)); break;
153 default: VC_CORE_ERROR_NO_EXIT("Unknown mouse button action: {}", action);
154 }
155 });

161 glfwSetCursorPosCallback(m_window, [](GLFWwindow* window, double xPos, double yPos) {
162 WindowData& data = *static_cast<WindowData*>(glfwGetWindowUserPointer(window));
163 const double dx = data.hasCursorPosition ? xPos - data.cursorX : 0.0;
164 const double dy = data.hasCursorPosition ? yPos - data.cursorY : 0.0;
165 data.cursorX = xPos;
166 data.cursorY = yPos;
167 data.hasCursorPosition = true;
168 postEvent<MouseMovedEvent>(window, static_cast<float>(xPos), static_cast<float>(yPos),
169 static_cast<float>(dx), static_cast<float>(dy));
170 });
171
172 glfwSetCursorEnterCallback(m_window, [](GLFWwindow* window, int entered) {
173 if (entered == GLFW_FALSE)
174 static_cast<WindowData*>(glfwGetWindowUserPointer(window))->hasCursorPosition = false;
175 postEvent<MouseEnterEvent>(window, entered == GLFW_TRUE);
176 });

217 DisplayServer Window::detectLinuxDisplayServer() {
218 const char* sessionType = std::getenv("XDG_SESSION_TYPE");
219
220 if (sessionType != nullptr) {
221 std::string type(sessionType);
222 if (type == "wayland") return WAYLAND;
223
224 if (type == "x11") return X11;
225
226 VC_CORE_ERROR_NO_EXIT("Unknown XDG_SESSION_TYPE variable : {}", sessionType);
227 return UNKNOWN_DISPLAY_SERVER;
228 }
229
230 VC_CORE_ERROR_NO_EXIT("Undefined XDG_SESSION_TYPE variable");
231 return UNKNOWN_DISPLAY_SERVER;
232 }
