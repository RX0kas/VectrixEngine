54 std::pair<VectrixResult, std::string> ProjectSerializer::createProject(const std::filesystem::path& directory, const std::string& name) {
55 VC_CORE_INFO("Creating project \"{}\" in {}", name.c_str(), toUtf8(directory).c_str());
56
57 // The name becomes the .vcproj and starting scene file names: it must not be empty or leave the project folder
58 if (name.empty() || name == "." || name == ".." || name.find_first_of("/\\") != std::string::npos) {
59 VC_CORE_ERROR_NO_EXIT("Invalid project name \"{}\"", name.c_str());
60 return {WRONG_FILE, "The project name can't be empty or contain '/' or '\\'"};
61 }
62
63 std::error_code ec;
64 if (std::filesystem::exists(directory, ec) && !std::filesystem::is_empty(directory, ec)) {
65 VC_CORE_WARN("Can't create project \"{}\": a project already exists at {}", name.c_str(), toUtf8(directory).c_str());
66 return {ALREADY_EXISTS, "A project already exists at " + toUtf8(directory)};
67 }
68
69 if (!std::filesystem::create_directories(directory, ec) && ec) {
70 VC_CORE_ERROR_NO_EXIT("Can't create project directory: {}", toUtf8(directory).c_str());
71 return {UNKNOWN_ERROR, "Can't create the project directory: " + ec.message()};
72 }
73
74 createProjectFolders(directory);
75
76 const std::filesystem::path startScene = std::filesystem::path("Scenes") / fromUtf8(name + ".vctx");
77 Scene scene(name);
78 if (SceneSerializer::saveScene(toUtf8(directory / startScene), scene) != SUCCESS) {
79 return {UNKNOWN_ERROR, "Can't write the starting scene of the project"};
80 }
81
82 std::ofstream file(directory / fromUtf8(name + ".vcproj"), std::ios::binary);
83 if (!file) {
84 VC_CORE_ERROR_NO_EXIT("Can't create project file in: {}", toUtf8(directory).c_str());
85 return {UNKNOWN_ERROR, "Can't create the .vcProj file"};
86 }
87
88 const uint32_t engineVersion = ApplicationInfo::getEngineVersion();
89 file.write(reinterpret_cast<const char*>(&MAGIC_NUMBER), sizeof(MAGIC_NUMBER));
90 file.write(reinterpret_cast<const char*>(&engineVersion), sizeof(engineVersion));
91 file.write(reinterpret_cast<const char*>(&PROJECT_VERSION), sizeof(PROJECT_VERSION));
92 writeString(file, name);
93 writeString(file, toGenericUtf8(startScene));
94 file.close();
95 if (!file) {
96 VC_CORE_ERROR_NO_EXIT("Failed while writing the project file in: {}", toUtf8(directory).c_str());
97 return {UNKNOWN_ERROR, "Can't write the .vcProj file"};
98 }
99
100 if (Json::save(toUtf8(directory / settingsFileName), JsonObject{}) != SUCCESS) {
101 return {UNKNOWN_ERROR, "Can't create the project settings file"};
102 }
103
104 VC_CORE_INFO("Project \"{}\" created successfully in {}", name.c_str(), toUtf8(directory).c_str());
105 return {SUCCESS, {}};
106 }

108 ProjectLoadResult ProjectSerializer::loadProject(const std::filesystem::path& path) {
109 VC_CORE_INFO("Loading project from {}", toUtf8(path).c_str());
110
111 std::ifstream file(path, std::ios::binary);
112 if (!file) {
113 VC_CORE_ERROR_NO_EXIT("Can't find file: {}", toUtf8(path).c_str());
114 return {.result = NOT_FOUND};
115 }
116
117 if (!validMagicNumber(file)) {
118 VC_CORE_ERROR_NO_EXIT("The file {} is not a Vectrix project file", toUtf8(path).c_str());
119 return {.result = WRONG_FILE};
120 }
121
122 if (!validVectrixVersion(file).has_value()) {
123 VC_CORE_ERROR_NO_EXIT("The file {} is not made for this Vectrix version", toUtf8(path).c_str());
124 return {.result = OUTDATED};
125 }
126
127 if (!validProjectVersion(file).has_value()) {
128 VC_CORE_ERROR_NO_EXIT("The file {} is too old", toUtf8(path).c_str());
129 return {.result = OUTDATED};
130 }
131
132 const std::optional<std::string> name = getString(file);
133 if (!name.has_value()) {
134 VC_CORE_ERROR_NO_EXIT("Can't load the name of the project from file: {}", toUtf8(path).c_str());
135 return {.result = UNKNOWN_ERROR};
136 }
137
138 const std::optional<std::string> startScene = getString(file);
139 if (!startScene.has_value()) {
140 VC_CORE_ERROR_NO_EXIT("Can't load the starting scene of the project from file: {}", toUtf8(path).c_str());
141 return {.result = UNKNOWN_ERROR};
142 }
143
144 VC_CORE_INFO("Project \"{}\" loaded successfully, starting scene: {}", name->c_str(), startScene->c_str());
145 return {.result = SUCCESS, .name = name.value(), .startScenePath = fromUtf8(startScene.value())};
146 }
