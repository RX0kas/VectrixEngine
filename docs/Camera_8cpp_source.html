59 void Camera::setViewDirection(glm::vec3 direction) {
60 VC_PROFILER_FUNCTION();
61 const glm::vec3 up = {0.0f, 1.0f, 0.0f};
62 const glm::vec3 w{glm::normalize(direction)};

64 const glm::vec3 v{glm::cross(w, u)};
65
66 m_viewMatrix = glm::mat4{1.f};
67 m_viewMatrix[0][0] = u.x;
68 m_viewMatrix[1][0] = u.y;

70 m_viewMatrix[0][1] = v.x;
71 m_viewMatrix[1][1] = v.y;
72 m_viewMatrix[2][1] = v.z;
73 m_viewMatrix[0][2] = w.x;
74 m_viewMatrix[1][2] = w.y;
75 m_viewMatrix[2][2] = w.z;
76 const glm::vec3 position = transform().position;
77 m_viewMatrix[3][0] = -glm::dot(u, position);
78 m_viewMatrix[3][1] = -glm::dot(v, position);
79 m_viewMatrix[3][2] = -glm::dot(w, position);
