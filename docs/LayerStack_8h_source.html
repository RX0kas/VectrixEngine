16 /**
17 * @brief Holds the layers of the application, in the order they were pushed
18 *
19 * The stack is split in two groups: the layers, then the overlays above them. Updates and renders go from the
20 * bottom up (the layers in the order they were pushed, then the overlays), so what is pushed later draws on
21 * top; events go from the top down (the last overlay pushed first, the first layer pushed last), so what is
22 * on top can handle an event before what is under it. Names don't have to be unique.
23 * @see Layer
24 * @see Application::PushLayer
25 * @ingroup layers
26 */
27 class LayerStack {
28 public:
29 LayerStack();
30 ~LayerStack();
31
32 /**
33 * @brief Add a layer, on top of the layers already there but under the overlays
34 * @param layer The layer to add, the stack shares its ownership
35 * @note Layer::OnAttach is called by the Application, not by the stack. A layer already in the stack isn't
36 * added again
37 */
38 void PushLayer(const std::shared_ptr<Layer>& layer);
39
40 /**
41 * @brief Add an overlay, on top of everything else
42 * @param overlay The overlay to add, the stack shares its ownership
43 * @see PushLayer
44 */
45 void PushOverlay(const std::shared_ptr<Layer>& overlay);
46
47 /**
48 * @brief Take a layer out of the stack
49 * @param layer The name of the layer to remove: the lowest layer with that name, else overlay
50 * @return The removed layer, or nullptr when no layer has that name
51 * @note Layer::OnDetach is called by the Application, not by the stack
52 */
53 std::shared_ptr<Layer> PopLayer(const std::string& layer);
54
55 /**
56 * @brief Put a layer in the place of another, so the order of the others doesn't change
57 * @param oldLayer The layer to replace
58 * @param newLayer The layer taking its place (on top of the layers when oldLayer isn't in the stack)
59 * @return The replaced layer, or nullptr when oldLayer isn't in the stack
60 * @note Layer::OnDetach and OnAttach are called by the Application
61 * @see Application::switchToLayer
62 */
63 std::shared_ptr<Layer> ReplaceLayer(const Layer* oldLayer, std::shared_ptr<Layer> newLayer);
64
65 /// The layers, from the bottom (pushed first) to the top
66 [[nodiscard]] const std::vector<std::shared_ptr<Layer>>& layers() const { return m_layers; }
67 /// The overlays, from the bottom (pushed first) to the top
68 [[nodiscard]] const std::vector<std::shared_ptr<Layer>>& overlays() const { return m_overlays; }
69
70 /**
71 * @brief Drop every layer and overlay of the stack
72 * @post The stack is empty
73 * @note Layer::OnDetach is called by the Application beforehand
74 */
75 void destroy();
76 private:
77 /// @return true when layer is in the stack already
78 [[nodiscard]] bool contains(const Layer* layer) const;
79
80 std::vector<std::shared_ptr<Layer>> m_layers;
81 std::vector<std::shared_ptr<Layer>> m_overlays;
82 };
