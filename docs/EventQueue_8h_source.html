21 /**
22 * @brief Holds the events posted during a frame until they are sent, at the start of the next one
23 *
24 * The Application owns one: the window posts the input to it, and Application::postEvent posts there too.
25 * On the thread that created the queue (the main thread), the events are built in a buffer reused from
26 * frame to frame, so posting doesn't allocate once the buffer has grown to what a frame needs.
27 *
28 * Any other thread can post too: its events are allocated apart, behind a lock, and sent after the main
29 * thread's ones. The order is kept per thread, not between threads.
30 * @note dispatch is main thread only
31 * @ingroup events
32 */
33 class EventQueue {
34 public:
35 EventQueue() = default;
36 ~EventQueue() {
37 m_batches[0].clear();
38 m_batches[1].clear();
39 }

43 /**
44 * @brief Build a T event in the queue
45 * @param args The arguments of T's constructor
46 */
47 template<std::derived_from<Event> T, typename... Args> requires std::constructible_from<T, Args...>
48 void post(Args&&... args) {
49 static_assert(alignof(T) <= alignof(std::max_align_t), "Over-aligned events aren't supported");
50 if (std::this_thread::get_id() != m_mainThread) {
51 auto event = std::make_unique<T>(std::forward<Args>(args)...); // built outside the lock
52 const std::scoped_lock lock(m_inboxMutex);
53 m_inbox.push_back(std::move(event));
54 return;
55 }
56 Batch& batch = m_batches[m_writing];
57 void* memory = batch.arena.allocate(sizeof(T), alignof(T));
58 batch.events.push_back(new (memory) T(std::forward<Args>(args)...));
59 }
60
61 /**
62 * @brief Give every event posted so far to a function, in the order they were posted, then drop them
63 *
64 * The events posted meanwhile (by the function) wait for the next call.
65 * @param send Called with each event
66 */
67 template<std::invocable<Event&> Fn>
68 void dispatch(Fn&& send) {
69 Batch& batch = m_batches[m_writing];
70 m_writing ^= 1;
71 {
72 // Swapped with a vector that keeps its capacity: no allocation in the lock in the usual case
73 const std::scoped_lock lock(m_inboxMutex);
74 m_inbox.swap(m_inboxSending);
75 }
76 for (Event* event : batch.events)
77 send(*event);
78 for (const std::unique_ptr<Event>& event : m_inboxSending)
79 send(*event);
80 batch.clear();
81 m_inboxSending.clear();
82 }
83
84 /// @brief Tell if no event is waiting
85 [[nodiscard]] bool empty() const {
86 if (!m_batches[m_writing].events.empty())
87 return false;
88 const std::scoped_lock lock(m_inboxMutex);

92 // A bump allocator: what it gives back is freed all at once by reset. Past its capacity it allocates
93 // apart, and grows by that much at the next reset
94 class Arena {
95 public:
96 void* allocate(size_t size, size_t alignment) {
97 if (!m_buffer) {
98 m_buffer = std::make_unique<std::byte[]>(m_capacity);
99 }
100 const size_t offset = (m_used + alignment - 1) & ~(alignment - 1);
101 if (offset + size > m_capacity) {
102 m_overflowSize += size + alignment;
103 return m_overflow.emplace_back(std::make_unique<std::byte[]>(size)).get();
104 }
105 m_used = offset + size;
106 return m_buffer.get() + offset;
107 }
108
109 void reset() {
110 if (!m_overflow.empty()) {
111 m_capacity += m_overflowSize;
112 m_buffer.reset(); // allocated again, bigger, on next use
113 m_overflow.clear();
114 m_overflowSize = 0;
115 }
116 m_used = 0;
117 }
118 private:
119 static constexpr size_t k_initialCapacity = 16 * 1024;
120
121 std::unique_ptr<std::byte[]> m_buffer;
122 size_t m_capacity = k_initialCapacity;
123 size_t m_used = 0;
124 std::vector<std::unique_ptr<std::byte[]>> m_overflow;
125 size_t m_overflowSize = 0;
126 };
