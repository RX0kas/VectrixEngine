16 /// Bytes left between the read position and the end of the file: sizes read from the file are
17 /// checked against it before allocating, so a corrupt size can't ask for gigabytes
18 std::uint64_t remainingBytes(std::ifstream& stream) {
19 const std::streampos position = stream.tellg();
20 stream.seekg(0, std::ios::end);
21 const std::streampos end = stream.tellg();
22 stream.seekg(position);
23 return end > position ? static_cast<std::uint64_t>(end - position) : 0;
24 }

32 void appendString(std::vector<std::byte>& bytes, const std::string& str) {
33 const auto len = static_cast<uint16_t>(str.size());
34 const auto* lenBytes = reinterpret_cast<const std::byte*>(&len);
35 bytes.insert(bytes.end(), lenBytes, lenBytes + sizeof(len));
36 const auto* strBytes = reinterpret_cast<const std::byte*>(str.data());
37 bytes.insert(bytes.end(), strBytes, strBytes + len);
38 }

40 /// A MeshRendererComponent as stored: its mesh, texture and shader ids (each a uint16 length then the bytes),
41 /// then the enable flag. Returns the component with the ids remap changes, nullopt when it changes none
42 std::optional<std::vector<std::byte>> remapMeshRendererIds(const std::vector<std::byte>& data, const std::function<std::optional<std::string>(const std::string&)>& remap, bool& malformed) {
43 std::vector<std::byte> rebuilt;
44 size_t offset = 0;
45 bool changed = false;
46 for (int field = 0; field < 3; ++field) {
47 std::uint16_t length = 0;
48 if (data.size() - offset < sizeof(length)) { malformed = true; return std::nullopt; }
49 std::memcpy(&length, data.data() + offset, sizeof(length));
50 offset += sizeof(length);
51 if (data.size() - offset < length) { malformed = true; return std::nullopt; }
52 std::string id(reinterpret_cast<const char*>(data.data() + offset), length);
53 offset += length;
54
55 if (std::optional<std::string> moved = id.empty() ? std::nullopt : remap(id)) {
56 id = std::move(*moved);
57 changed = true;
58 }
59 appendString(rebuilt, id);
60 }
61 if (!changed)
62 return std::nullopt;
63 rebuilt.insert(rebuilt.end(), data.begin() + static_cast<std::ptrdiff_t>(offset), data.end());
64 return rebuilt;
65 }

69 SceneCreationData SceneSerializer::loadSceneFile(const std::string& path) {
70 std::ifstream file(fromUtf8(path),std::ios::binary);
71 if (!file) {
72 VC_CORE_ERROR_NO_EXIT("Can't find file: {}",path.c_str());
73 return {.result = NOT_FOUND};
74 }
75
76 if (!validMagicNumber(file)) {
77 VC_CORE_ERROR_NO_EXIT("The file {} is not a Vectrix scene file",path.c_str());
78 return {.result = WRONG_FILE};
79 }
80
81 std::optional<uint32_t> vectrixVersion = validVectrixVersion(file);
82
83 if (!vectrixVersion.has_value()) {
84 VC_CORE_ERROR_NO_EXIT("The file {} is not made for this Vectrix version",path.c_str());
85 return {.result = OUTDATED};
86 }
87
88 std::optional<uint32_t> sceneVersion = validSceneVersion(file);
89 if (!sceneVersion.has_value()) {
90 VC_CORE_ERROR_NO_EXIT("The file {} is too old",path.c_str());
91 return {.result = OUTDATED};
92 }
93
94
95 SceneCreationData data{};
96 data.file_version = sceneVersion.value();
97 data.engine_version = vectrixVersion.value();
98
99 std::optional<std::string> sceneName = getString(file);
100
101 if (!sceneName.has_value()) {
102 VC_CORE_ERROR_NO_EXIT("Can't load the name of the scene from file: {}",path.c_str());
103 return {.result = UNKNOWN_ERROR};
104 }
105 data.name = sceneName.value();
106
107 auto entities = readEntities(file);
108 if (!entities.has_value()) {
109 VC_CORE_ERROR_NO_EXIT("Can't load the entities of the scene from file: {}",path.c_str());
110 return {.result = UNKNOWN_ERROR};
111 }
112 data.entities = entities.value();
113
114 data.result = SUCCESS;
115 return data;
116 }

152 std::optional<std::vector<EntityCreationData>> SceneSerializer::readEntities(std::ifstream &stream) {
153 std::uint32_t entityCount;
154 if (!stream.read(reinterpret_cast<char*>(&entityCount), sizeof(entityCount))) {
155 return std::nullopt;
156 }
157
158 // Every entity takes at least its name length and component count (4 bytes)
159 if (entityCount > remainingBytes(stream) / 4) {
160 return std::nullopt;
161 }
162
163 std::vector<EntityCreationData> datas(entityCount);
164 for (std::uint32_t entitiesLoaded = 1; entitiesLoaded <= entityCount; entitiesLoaded++) {
165 std::optional<std::string> entityName = getString(stream);
166
167 if (!entityName.has_value()) {
168 return std::nullopt;
169 }
170
171 std::optional<std::vector<ComponentCreationData>> components = readComponents(stream);
172 if (!components.has_value()) {
173 return std::nullopt;
174 }
175
176 EntityCreationData data{};
177 data.name = entityName.value();
178 data.components = components.value();
179
180 datas[entitiesLoaded-1] = data;
181 }
182
183 return datas;
184 }

186 std::optional<std::vector<ComponentCreationData>> SceneSerializer::readComponents(std::ifstream &stream) {
187 std::uint16_t componentsCount;
188 if (!stream.read(reinterpret_cast<char*>(&componentsCount), sizeof(componentsCount))) {
189 return std::nullopt;
190 }
191
192
193 std::vector<ComponentCreationData> datas;
194 for (std::uint16_t componentLoaded = 0; componentLoaded < componentsCount; componentLoaded++) {
195 std::uint32_t type_id;
196 if (!stream.read(reinterpret_cast<char*>(&type_id), sizeof(type_id))) {
197 return std::nullopt;
198 }
199
200 std::uint32_t data_size;
201 if (!stream.read(reinterpret_cast<char*>(&data_size), sizeof(data_size))) {
202 return std::nullopt;
203 }
204
205 if (data_size > remainingBytes(stream)) {
206 return std::nullopt;
207 }
208
209 std::vector<std::byte> data(data_size);
210 if (!stream.read(reinterpret_cast<char*>(data.data()), data_size)) {
211 return std::nullopt;
212 }
213
214 ComponentCreationData creationData;
215 creationData.type_id = type_id;
216 creationData.data = data;
217
218 datas.push_back(creationData);
219 }
220
221 return datas;
222 }

224 std::pair<VectrixResult, bool> SceneSerializer::remapAssetIds(const std::string& path, const std::function<std::optional<std::string>(const std::string&)>& remap) {
225 SceneCreationData data = loadSceneFile(path);
226 if (data.result != SUCCESS)
227 return {data.result, false};
228
229 bool changed = false;
230 const std::uint32_t meshRendererType = entt::type_hash<MeshRendererComponent>::value();
231 for (EntityCreationData& entity : data.entities) {
232 for (ComponentCreationData& component : entity.components) {
233 if (component.type_id != meshRendererType)
234 continue;
235 bool malformed = false;
236 if (std::optional<std::vector<std::byte>> remapped = remapMeshRendererIds(component.data, remap, malformed)) {
237 component.data = std::move(*remapped);
238 changed = true;
239 } else if (malformed) {
240 VC_CORE_ERROR_NO_EXIT("Malformed Mesh Renderer in entity {} of scene {}", entity.name, path);
241 return {FORMATING_ERROR, false};
242 }
243 }
244 }
245 if (!changed)
246 return {SUCCESS, false};
247
248 // Everything else is written back as it was read, versions included
249 const std::filesystem::path file = fromUtf8(path);
250 std::filesystem::path temporary = file;
251 temporary += ".tmp";
252 {
253 std::ofstream out(temporary, std::ios::binary | std::ios::trunc);
254 out.write(reinterpret_cast<const char*>(&MAGIC_NUMBER), sizeof(MAGIC_NUMBER));
255 out.write(reinterpret_cast<const char*>(&data.engine_version), sizeof(data.engine_version));
256 out.write(reinterpret_cast<const char*>(&data.file_version), sizeof(data.file_version));
257 writeString(out, data.name);
258
259 const auto entityCount = static_cast<std::uint32_t>(data.entities.size());
260 out.write(reinterpret_cast<const char*>(&entityCount), sizeof(entityCount));
261 for (const EntityCreationData& entity : data.entities) {
262 writeString(out, entity.name);
263 const auto componentCount = static_cast<std::uint16_t>(entity.components.size());
264 out.write(reinterpret_cast<const char*>(&componentCount), sizeof(componentCount));
265 for (const ComponentCreationData& component : entity.components) {
266 const auto size = static_cast<std::uint32_t>(component.data.size());
267 out.write(reinterpret_cast<const char*>(&component.type_id), sizeof(component.type_id));
268 out.write(reinterpret_cast<const char*>(&size), sizeof(size));
269 out.write(reinterpret_cast<const char*>(component.data.data()), size);
270 }
271 }
272 if (!out.good()) {
273 VC_CORE_ERROR_NO_EXIT("Failed while writing: {}", toUtf8(temporary));
274 out.close();
275 std::error_code ignored;
276 std::filesystem::remove(temporary, ignored);
277 return {UNKNOWN_ERROR, false};
278 }
279 }
280
281 std::error_code ec;
282 std::filesystem::rename(temporary, file, ec); // replaces the scene in one step
283 if (ec) {
284 VC_CORE_ERROR_NO_EXIT("Can't replace {}: {}", path, ec.message());
285 std::filesystem::remove(temporary, ec);
286 return {UNKNOWN_ERROR, false};
287 }
288 return {SUCCESS, true};
289 }

291 VectrixResult SceneSerializer::saveScene(const std::string &path, Scene& scene) {
292 std::ofstream file(fromUtf8(path), std::ios::binary);
293 if (!file) {
294 VC_CORE_ERROR_NO_EXIT("Can't save file: {}", path.c_str());
295 return UNKNOWN_ERROR;
296 }
297
298 // Header
299 uint32_t engineVersion = ApplicationInfo::getEngineVersion();
300 file.write(reinterpret_cast<const char*>(&MAGIC_NUMBER), sizeof(MAGIC_NUMBER));
301 file.write(reinterpret_cast<const char*>(&engineVersion), sizeof(engineVersion));
302 file.write(reinterpret_cast<const char*>(&SCENE_VERSION), sizeof(SCENE_VERSION));
303
304 // Name
305 writeString(file,scene.getName());
306
307 auto view = scene.m_registry.view<InformationComponent>();
308 uint32_t entityCount = view.size();
309 file.write(reinterpret_cast<const char*>(&entityCount), sizeof(entityCount));
310 for (const entt::entity& entity : view) {
311 std::shared_ptr<Entity> e = scene.getEntity(entity);
312 writeString(file,e->getComponent<InformationComponent>().name);
313
314 uint16_t componentCount = 1;
315 if (e->hasComponent<CameraComponent>()) componentCount++;
316 if (e->hasComponent<MeshRendererComponent>()) componentCount++;
317 file.write(reinterpret_cast<const char*>(&componentCount), sizeof(componentCount));
318
319 // Components
320 // TransformComponent
321 TransformComponent transform = e->getComponent<TransformComponent>();
322 uint32_t hashTransform = entt::type_hash<TransformComponent>::value();
323 uint32_t transformSize = sizeof(TransformComponent);
324 file.write(reinterpret_cast<const char*>(&hashTransform), sizeof(hashTransform));
325 file.write(reinterpret_cast<const char*>(&transformSize), sizeof(transformSize));
326 file.write(reinterpret_cast<const char*>(&transform), sizeof(transform));
327
328 if (e->hasComponent<CameraComponent>()) {
329 CameraComponent camera = e->getComponent<CameraComponent>();
330 uint32_t hashCamera = entt::type_hash<CameraComponent>::value();
331 uint32_t cameraComponentSize = sizeof(float)*4;
332 file.write(reinterpret_cast<const char*>(&hashCamera), sizeof(hashCamera));
333 file.write(reinterpret_cast<const char*>(&cameraComponentSize), sizeof(cameraComponentSize));
334
335 // Data
336 float fov = camera.camera.getFOV();
337 file.write(reinterpret_cast<const char*>(&fov), sizeof(float));
338
339 float camNear = camera.camera.getCamNear();
340 file.write(reinterpret_cast<const char*>(&camNear), sizeof(float));
341
342 float camFar = camera.camera.getCamFar();
343 file.write(reinterpret_cast<const char*>(&camFar), sizeof(float));
344
345 const float aspect = camera.camera.hasCustomAspect() ? camera.camera.getAspect() : -1.0f;
346 file.write(reinterpret_cast<const char*>(&aspect), sizeof(float));
347 }
348
349 if (e->hasComponent<MeshRendererComponent>()) {
350 auto& mesh = e->getComponent<MeshRendererComponent>();
351 uint32_t hashMesh = entt::type_hash<MeshRendererComponent>::value();
352
353 // The editor lets a component be saved before all its assets are set: an unset asset is an empty id
354 // A missing asset is saved as the path it had, so the reference isn't lost (for a texture, rather
355 // than the not_found texture standing in for it)
356 const auto idOf = [](const std::string& missingId, const auto& asset) {
357 return !missingId.empty() ? missingId : asset ? asset->getID() : std::string();
358 };
359 const std::string meshId = idOf(mesh.missingMesh, mesh.mesh);
360 const std::string textureId = idOf(mesh.missingTexture, mesh.texture);
361 const std::string shaderId = idOf(mesh.missingShader, mesh.shader);
362
363 // After the flag, each asset's fingerprint, to find it again if it's moved outside the editor. A missing
364 // one keeps the fingerprint it was loaded with
365 const auto fingerprintOf = [&scene](const std::string& id, const std::string& missingId) {
366 if (id.empty())
367 return AssetFingerprint{};
368 if (!missingId.empty()) {
369 const auto it = scene.m_missingFingerprints.find(missingId);
370 return it != scene.m_missingFingerprints.end() ? it->second : AssetFingerprint{};
371 }
372 return AssetsManager::getFingerprint(id).value_or(AssetFingerprint{});
373 };
374 const std::array<AssetFingerprint, 3> fingerprints = {
375 fingerprintOf(meshId, mesh.missingMesh), fingerprintOf(textureId, mesh.missingTexture), fingerprintOf(shaderId, mesh.missingShader)
376 };
377
378 uint32_t meshSize = sizeof(uint16_t) + meshId.size()
379 + sizeof(uint16_t) + textureId.size()
380 + sizeof(uint16_t) + shaderId.size() + sizeof(bool)
381 + static_cast<uint32_t>(fingerprints.size() * 2 * sizeof(std::uint64_t));
382 file.write(reinterpret_cast<const char*>(&hashMesh), sizeof(hashMesh));
383 file.write(reinterpret_cast<const char*>(&meshSize), sizeof(meshSize));
384 writeString(file, meshId);
385 writeString(file, textureId);
386 writeString(file, shaderId);
387 // Kept disabled only because of a missing asset: saved as it was, so it is drawn again once that's back
388 bool isEnable = mesh.isEnable() || (mesh.enabledOnceComplete && mesh.hasMissingAsset());
389 file.write(reinterpret_cast<const char*>(&isEnable),sizeof(bool));
390 for (const AssetFingerprint& fingerprint : fingerprints) {
391 file.write(reinterpret_cast<const char*>(&fingerprint.size), sizeof(fingerprint.size));
392 file.write(reinterpret_cast<const char*>(&fingerprint.hash), sizeof(fingerprint.hash));
393 }
394 }
395 }
396
397 if (!file.good()) {
398 VC_CORE_ERROR_NO_EXIT("Failed while writing: {}", path.c_str());
399 return UNKNOWN_ERROR;
400 }
401 return SUCCESS;
402 }
