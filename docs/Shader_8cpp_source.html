69 ShaderUniformLayout Shader::findShaderUniformLayoutFromSource(const std::string &source, bool &isAffectedByCamera) {
70 std::istringstream file(source);
71 ShaderUniformLayout layout;
72 std::string line;
73 std::stringstream contentStream;
74 bool isContent = false;
75 isAffectedByCamera = false;
76
77 while (std::getline(file, line)) {
78 std::string lineCopy = normalizeSpaces(line);
79 std::string lineWithOutSpace = lineCopy;
80 replaceAll(lineWithOutSpace," ","");
81 // Trim leading space
82 if (!lineCopy.empty() && lineCopy.front() == ' ')
83 lineCopy = lineCopy.substr(1);
84
85 if (lineWithOutSpace.find("layout(set=0,binding=0)readonlybuffer") != std::string::npos) {
86 isContent = true;
87 continue;
88 }
89
90
91 if (isContent && lineCopy.find('{') != std::string::npos) {
92 continue;
93 }
94 if (isContent && lineCopy.find('}') != std::string::npos) {
95 break;
96 }
97 if (isContent) {
98 if (lineCopy.find("vc_cameraTransform") != std::string::npos)
99 isAffectedByCamera = true;
100 contentStream << lineCopy << ' ';
101 }
102 }
103
104 std::string content = contentStream.str();
105 std::stringstream layoutStream;
106 ShaderUniformType type = ShaderUniformType::Float;
107 bool lastWasSpace = false;
108
109 for (char c : content) {
110 if (c == '\r') { continue; }
111
112 if (c == ';') {
113 std::string name = layoutStream.str();
114 replaceAll(name, " ", "");
115 if (!name.empty()) {
116 layout.add(name, type);
117 }
118 layoutStream.str("");
119 layoutStream.clear();
120 lastWasSpace = false;
121 continue;
122 }
123
124 if (c == ' ' && !lastWasSpace) {
125 std::string t = layoutStream.str();
126 replaceAll(t, " ", "");
127 if (t == "int") {
128 type = ShaderUniformType::Int;
129 }
130 else if (t == "float") {
131 type = ShaderUniformType::Float;
132 }
133 else if (t == "vec2") {
134 type = ShaderUniformType::Vec2;
135 }
136 else if (t == "vec3") {
137 type = ShaderUniformType::Vec3;
138 }
139 else if (t == "vec4") {
140 type = ShaderUniformType::Vec4;
141 }
142 else if (t == "mat4") {
143 type = ShaderUniformType::Mat4;
144 }
145 else if (t == "uint") {
146 type = ShaderUniformType::Uint;
147 }
148 else if (t == "bool") {
149 type = ShaderUniformType::Bool;
150 }
151 else if (!t.empty()) {
152 VC_CORE_ERROR("Unknown type {}", t);
153 }
154 layoutStream.str("");
155 layoutStream.clear();
156 lastWasSpace = true;
157 continue;
158 }
159
160 if (c == ' ' && lastWasSpace)
161 continue;
162
163 lastWasSpace = false;
164 layoutStream << c;
165 }
166
167 return layout;
168 }
