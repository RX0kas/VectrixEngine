53 /**
54 * @brief Translate an engine format into the Vulkan one
55 * @param f The engine format
56 * @return The matching Vulkan format, `VK_FORMAT_UNDEFINED` when there is none
57 * @see toVectrixFormat
58 * @ingroup utils
59 */
60 inline VkFormat toVulkanFormat(VectrixImageFormat f) {
61 switch (f) {
62 case UNDEFINED: return VK_FORMAT_UNDEFINED;
63 case RGBA_UNORM: return VK_FORMAT_R8G8B8A8_UNORM;
64 case BGRA_SRGB: return VK_FORMAT_B8G8R8A8_SRGB;
65 case R_UNORM: return VK_FORMAT_R8_UNORM;
66 case R_SRGB: return VK_FORMAT_R8_SRGB;
67 case D32_SFLOAT: return VK_FORMAT_D32_SFLOAT;
68 case D32_SFLOAT_S8_UINT: return VK_FORMAT_D32_SFLOAT_S8_UINT;
69 case D24_UNORM_S8_UINT: return VK_FORMAT_D24_UNORM_S8_UINT;
70 default: return VK_FORMAT_UNDEFINED;
71 }
72 }
