19 /**
20 * @brief The families an Event can belong to
21 *
22 * The values are bit flags, combined with `|`: a KeyPressedEvent is both #Keyboard and #Input.
23 * @see Event::isInCategory
24 * @ingroup events
25 */
26 enum class EventCategory : uint8_t {
27 None = 0,
28 Window = BIT(0), ///< Sent by the window itself (resize, close, move, focus, files dropped)
29 Input = BIT(1), ///< Sent by any input device
30 Keyboard = BIT(2), ///< Sent by the keyboard (keys and typed text)
31 Mouse = BIT(3), ///< Sent by the mouse
32 MouseButton = BIT(4), ///< Sent by a mouse button specifically
33 };
34
35 /// @brief Combine two sets of categories
36 constexpr EventCategory operator|(EventCategory a, EventCategory b) {
37 return static_cast<EventCategory>(static_cast<uint8_t>(a) | static_cast<uint8_t>(b));
38 }
39 /// @brief Keep the categories two sets have in common
40 constexpr EventCategory operator&(EventCategory a, EventCategory b) {
41 return static_cast<EventCategory>(static_cast<uint8_t>(a) & static_cast<uint8_t>(b));
42 }

144 /**
145 * @brief The base of every concrete event: gives it its type id, name and categories
146 *
147 * Derive the event from it, naming the class itself, then add the event's data as public members:
148 * @code
149 * struct SceneOpenedEvent : Vectrix::EventBase<SceneOpenedEvent, "SceneOpened"> {
150 * explicit SceneOpenedEvent(std::string path) : path(std::move(path)) {}
151 * std::string path;
152 * };
153 *
154 * Application::instance().postEvent<SceneOpenedEvent>(scenePath);
155 * @endcode
156 * @tparam Derived The event class being defined
157 * @tparam Name The name getName returns, for logging
158 * @tparam Categories The categories of the event, combined with `|`
159 * @ingroup events
160 */
161 template<typename Derived, detail::EventName Name, EventCategory Categories = EventCategory::None>
162 class EventBase : public Event {
163 public:
164 /// @brief The categories every event of this class belongs to
165 static constexpr EventCategory staticCategories = Categories;
166
167 [[nodiscard]] std::string_view getName() const override { return Name.view(); }
168 protected:
169 /// @cond INTERNAL
170 EventBase() : Event(eventTypeId<Derived>(), Categories) {}
171 /// @endcond
172 };
173}
