25 /**
26 * @brief This function create a version from the different version number
27 * @param major A major version number
28 * @param minor A minor version number
29 * @param patch A patch version number
30 * @return The three numbers and the platform packed together
31 * @warning Major is stored on 7 bits (up to 127), minor and patch on 12 bits (up to 4095): a larger
32 * value is reported as an error and truncated, so it can't overflow into the other fields
33 */
34 static Version makeVersion(const std::uint32_t major, const std::uint32_t minor, const std::uint32_t patch) {
35 if (major>0x7FU) {
36 VC_CORE_ERROR_NO_EXIT("Major is too high to be fully registered");
37 }
38 if (minor>0xFFFU) {
39 VC_CORE_ERROR_NO_EXIT("Minor is too high to be fully registered");
40 }
41 if (patch>0xFFFU) {
42 VC_CORE_ERROR_NO_EXIT("Patch is too high to be fully registered");
43 }
44 return static_cast<uint32_t>(VC_PLATFORM_ID)<< 31U | (major & 0x7FU) << 24U | (minor & 0xFFFU) << 12U | (patch & 0xFFFU);
45 }

94 /**
95 * @brief Tell if something written with one version can be read by another
96 *
97 * The major numbers have to match, and the reader may not be older than the writer.
98 * @param version1 The version doing the reading
99 * @param version2 The version the content was written with
100 * @return true when version1 can handle what version2 produced
101 * @ingroup utils
102 */
103 static bool isCompatible(Version version1,Version version2) {
104 if (getMajor(version1)!=getMajor(version2)) return false;
105
106 if (getMinor(version1) < getMinor(version2)) return false;
107
108 return true;
109 }
