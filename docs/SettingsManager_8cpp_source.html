8 void mergeInto(JsonObject& base, const JsonObject& overlay) {
9 for (const auto& [key, overlayValue] : overlay) {
10 const auto baseIt = base.find(key);
11 if (baseIt != base.end() && baseIt->second.isType<JsonObject>() && overlayValue.isType<JsonObject>()) {
12 JsonObject merged = baseIt->second.getAsObject();
13 mergeInto(merged, overlayValue.getAsObject());
14 baseIt->second = JsonValue(merged);
15 } else {
16 base[key] = overlayValue;
17 }
18 }
19 }

48 std::pair<VectrixResult, std::string> SettingsManager::loadInto(JsonObject& target, const std::filesystem::path& file) {
49 if (!std::filesystem::exists(file)) {
50 target.clear();
51 return {SUCCESS, ""};
52 }
53
54 // On failure the tier is emptied rather than left holding what the previous file loaded into it
55 // (e.g. the previous project's settings when the new project's file is broken)
56 auto [result, root] = Json::load(toUtf8(file));
57 if (result != SUCCESS) {
58 target.clear();
59 return {result, "Error while parsing settings: " + root.getString()};
60 }
61 if (!root.isType<JsonObject>()) {
62 target.clear();
63 return {WRONG_TYPE, "Settings root is not a JSON object"};
64 }
65
66 target = root.getAsObject();
67 return {SUCCESS, ""};
68 }

76 std::pair<VectrixResult, std::string> SettingsManager::loadProject(const std::filesystem::path& file) {
77 m_effectiveDirty = true;
78
79 if (file.empty()) {
80 m_project.clear();
81 m_projectPath.clear();
82 return {SUCCESS, ""};
83 }
84
85 const auto outcome = loadInto(m_project, file);
86 m_projectPath = (outcome.first == SUCCESS) ? file : std::filesystem::path{};
87 return outcome;

90 std::pair<VectrixResult, std::string> SettingsManager::save(const Scope scope) const {
91 const std::filesystem::path& path = scope == Scope::Project ? m_projectPath : m_globalPath;
92 const JsonObject& data = scope == Scope::Project ? m_project : m_global;
93
