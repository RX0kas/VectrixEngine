268 inline std::pair<VectrixResult, std::shared_ptr<Texture>> AssetsManager::load(const std::string& path) {
269 const auto [p, id, inAssetsFolder] = resolve(path);
270
271 // Reported rather than replaced by the not_found texture: a caller like the scene loader has to know, to
272 // keep the reference (and it isn't cached, so the file loads once it's back)
273 if (!std::filesystem::exists(p))
274 return {VectrixResult::NOT_FOUND, nullptr};
275
276 if (getAssetType(p)!=TEXTURE) {
277 return {VectrixResult::WRONG_TYPE,nullptr};
278 }
279
280 const std::string cacheKey = toGenericUtf8(p);
281 auto it = s_instance->m_cache.find(cacheKey);
282 if (it != s_instance->m_cache.end()) {
283 auto texture = std::static_pointer_cast<Texture>(it->second);
284 if (s_instance->m_textureManager->m_cache.find(id) == s_instance->m_textureManager->m_cache.end())
285 s_instance->m_textureManager->add(id, texture);
286 return {VectrixResult::SUCCESS, texture};
287 }
288
289 auto texture = s_instance->m_textureManager->createTexture(id, toUtf8(p));
290 s_instance->m_cache.emplace(cacheKey, texture);
291 if (inAssetsFolder)
292 s_instance->m_projectAssets.push_back({TEXTURE, id, cacheKey});
293 return {VectrixResult::SUCCESS, texture};
294 }
