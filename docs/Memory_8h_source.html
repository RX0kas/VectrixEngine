30 /**
31 * @brief A hash map tuned for the engine, used to keep loaded assets around
32 *
33 * It is an `std::unordered_map` hashed with XXH3, pre reserved so the usual asset
34 * count does not cause a rehash. The managers use it to hand back an asset that was
35 * already loaded rather than loading it twice.
36 * @tparam K The key type, usually the name or the path of the asset
37 * @tparam I The stored type
38 * @see XXH3
39 * @ingroup utils
40 */
41 template<typename K,typename I>
42 class Cache : public std::unordered_map<K,I,XXH3> {
43 public:
44 /**
45 * @brief Build the cache, reserving room up front
46 * @param reservedN How many entries to make room for
47 * @param max_load_factor How full it may get before it rehashes
48 */
49 Cache(int reservedN = 256,float max_load_factor = 0.7f) {
50 this->reserve(reservedN);
51 this->max_load_factor(max_load_factor);
52 }

58 /**
59 * @brief Return every stored value, without its key
