107 /**
108 * @brief Receives the events it subscribed to
109 *
110 * Layer and Application are listeners: subscribe once, from the constructor or OnAttach, and the handler
111 * is called for every matching event until it is unsubscribed or the listener destroyed.
112 * @code
113 * MyLayer::MyLayer() : Layer("MyLayer") {
114 * subscribe<KeyPressedEvent>([this](const KeyPressedEvent& e) {
115 * if (e.key != VC_KEY_ESCAPE) return false; // let the layers below get it
116 * togglePause();
117 * return true; // consumed: the layers below don't get it
118 * });
119 * subscribe<WindowResizeEvent>(&MyLayer::onResize); // a member function works too
120 * subscribe<WindowFocusEvent>([this] { m_paused = true; }); // the event can be left out
121 * }
122 * @endcode
123 *
124 * The engine sends the events once per frame, before Layer::OnUpdate, from the top of the layer stack
125 * down (ImGui, the overlays from the last pushed, the layers from the last pushed) and then to the
126 * Application. Within a listener the handlers run in subscription order. Once a handler consumes the
127 * event, nothing after it receives it.
128 *
129 * Subscribing to Event itself (`subscribe<Event>`) receives every event.
130 * @note Handlers may subscribe and unsubscribe, even themselves, while an event is being sent.
131 * A handler added then only receives the next events
132 * @see Application::postEvent
133 * @ingroup events
134 */
135 class EventListener {
136 public:
137 EventListener() = default;
138 virtual ~EventListener() = default;

143 /**
144 * @brief Call a handler for every T event
145 * @tparam T The event class, or Event to receive every event
146 * @param handler A callable taking a `const T&`, or nothing, and returning bool (true consumes the
147 * event) or void (never consumes it)
148 * @return The id to give unsubscribe
149 */
150 template<std::derived_from<Event> T, typename F> requires EventHandler<std::decay_t<F>, T>
151 SubscriptionId subscribe(F&& handler) {
152 return add(eventTypeId<T>(), [h = std::forward<F>(handler)](const Event& event) mutable {
153 return invokeHandler(h, static_cast<const T&>(event));
154 });
155 }

157 /**
158 * @brief Like subscribe, but the subscription is removed when the returned handle is destroyed
159 *
160 * Meant for another object than the listener, which keeps the handle as a member.
161 * @tparam T The event class, or Event to receive every event
162 * @param handler A callable taking a `const T&`, or nothing, and returning bool or void
163 * @return The handle owning the subscription
164 */
165 template<std::derived_from<Event> T, typename F> requires EventHandler<std::decay_t<F>, T>
166 [[nodiscard]] ScopedSubscription subscribeScoped(F&& handler) {
167 return {m_lifetime, subscribe<T>(std::forward<F>(handler))};
168 }

224 template<typename T, typename F>
225 static bool invokeHandler(F& handler, const T& event) {
226 if constexpr (std::invocable<F&, const T&>) {
227 if constexpr (std::is_void_v<std::invoke_result_t<F&, const T&>>) {
228 std::invoke(handler, event);
229 return false;
230 } else {
231 return static_cast<bool>(std::invoke(handler, event));
232 }
233 } else {
234 if constexpr (std::is_void_v<std::invoke_result_t<F&>>) {
235 std::invoke(handler);
236 return false;
237 } else {
238 return static_cast<bool>(std::invoke(handler));
239 }
240 }
241 }

243 SubscriptionId add(EventTypeId type, std::function<bool(const Event&)> handler) {
244 const SubscriptionId id = m_nextId++;
245 m_subscriptions.push_back(std::make_unique<Subscription>(Subscription{id, type, std::move(handler)}));
246 return id;

249 // A handler that is running can't be destroyed: it is only deactivated, and erased once no event is
250 // being sent anymore
251 void remove(Subscription& subscription) {
252 subscription.active = false;
253 m_hasInactive = true;
254 }

256 void cleanUp() {
257 if (m_sending > 0 || !m_hasInactive) return;
258 std::erase_if(m_subscriptions, [](const auto& subscription) { return !subscription->active; });
259 m_hasInactive = false;
260 }

262 // Called by the Application for every event, see the class description for the order
263 void notify(Event& event) {
264 ++m_sending;
265 // The subscriptions are held by pointer and only counted up to here: one added by a handler moves the
266 // vector but not the subscription being run, and waits for the next event
267 const size_t count = m_subscriptions.size();
268 for (size_t i = 0; i < count && !event.m_handled; ++i) {
269 Subscription& subscription = *m_subscriptions[i];
270 if (!subscription.active) continue;
271 if (subscription.type == event.getType() || subscription.type == eventTypeId<Event>())
272 event.m_handled = subscription.handler(event);
273 }
274 --m_sending;
275 cleanUp();
276 }
