44 AssetsManager::ResolvedAsset AssetsManager::resolve(const std::string& path) {
45 // absolute(): the assets folder itself may be relative (a project path given on the command line)
46 std::error_code ec;
47 const std::filesystem::path root = std::filesystem::absolute(s_assetsPath, ec).lexically_normal();
48 const std::filesystem::path requested = fromUtf8(path);
49 const std::filesystem::path file = std::filesystem::absolute(requested.is_relative() ? s_assetsPath / requested : requested, ec).lexically_normal();
50
51 // Empty for another drive on Windows, starts with ".." for anything outside the folder
52 const std::filesystem::path relative = file.lexically_relative(root);
53 const bool insideAssets = !relative.empty() && *relative.begin() != "..";
54 return {file, insideAssets ? toGenericUtf8(relative) : toGenericUtf8(file), insideAssets};
55 }

83 void AssetsManager::moveProjectAssets(const std::string& oldId, const std::string& newId) {
84 for (ProjectAsset& asset : m_projectAssets) {
85 std::optional<std::string> movedId = remapAssetId(asset.id, oldId, newId);
86 if (!movedId)
87 continue;
88
89 std::shared_ptr<void> object;
90 if (const auto it = m_cache.find(asset.cacheKey); it != m_cache.end()) {
91 object = it->second;
92 m_cache.erase(it);
93 }
94 std::string cacheKey = toGenericUtf8(resolve(*movedId).file);
95 if (object)
96 m_cache.emplace(cacheKey, object);
97
98 // The type managers may not hold it anymore (EditorLayer::openScene clears the texture and mesh ones)
99 switch (object ? asset.type : UNKNOWN) {
100 case TEXTURE: {
101 const auto texture = std::static_pointer_cast<Texture>(object);
102 texture->setID(*movedId);
103 m_textureManager->remove(asset.id);
104 m_textureManager->add(*movedId, texture);
105 break;
106 }
107 case SHADER: {
108 const auto shader = std::static_pointer_cast<Shader>(object);
109 shader->setID(*movedId);
110 m_shaderManager->remove(asset.id);
111 m_shaderManager->add(*movedId, shader);
112 break;
113 }
114 case MESH: {
115 const auto mesh = std::static_pointer_cast<Mesh>(object);
116 mesh->m_id = *movedId;
117 m_meshManager->remove(asset.id);
118 m_meshManager->add(*movedId, mesh);
119 break;
120 }
121 default: break;
122 }
123 VC_CORE_INFO("Asset {} moved to {}", asset.id, *movedId);
124 asset.id = std::move(*movedId);
125 asset.cacheKey = std::move(cacheKey);
126 }
127 }

129 std::optional<AssetFingerprint> AssetsManager::getFingerprint(const std::string& id) {
130 const std::filesystem::path file = resolve(id).file;
131 std::error_code ec;
132 const std::uint64_t size = std::filesystem::file_size(file, ec);
133 if (ec)
134 return std::nullopt;
135 const std::filesystem::file_time_type writeTime = std::filesystem::last_write_time(file, ec);
136 if (ec)
137 return std::nullopt;
138
139 const std::string key = toGenericUtf8(file);
140 if (const auto it = s_instance->m_fingerprints.find(key); it != s_instance->m_fingerprints.end()
141 && it->second.size == size && it->second.writeTime == writeTime)
142 return AssetFingerprint{size, it->second.hash};
143
144 std::ifstream stream(file, std::ios::binary);
145 if (!stream)
146 return std::nullopt;
147 XXH3_state_t* state = XXH3_createState();
148 XXH3_64bits_reset_withSeed(state, XXH3::seed);
149 std::vector<char> buffer(1 << 16);
150 while (stream) {
151 stream.read(buffer.data(), static_cast<std::streamsize>(buffer.size()));
152 if (stream.gcount() > 0)
153 XXH3_64bits_update(state, buffer.data(), static_cast<size_t>(stream.gcount()));
154 }
155 const std::uint64_t hash = XXH3_64bits_digest(state);
156 XXH3_freeState(state);
157
158 s_instance->m_fingerprints[key] = {size, writeTime, hash};
159 return AssetFingerprint{size, hash};
160 }

162 std::optional<AssetsManager::MovedAsset> AssetsManager::findMovedAsset(const std::string& id, const AssetFingerprint& fingerprint) {
163 const std::filesystem::path missing = fromUtf8(id);
164 std::vector<std::filesystem::path> sameContent;
165 std::vector<std::filesystem::path> sameName;
166
167 std::error_code ec;
168 for (auto it = std::filesystem::recursive_directory_iterator(s_assetsPath, std::filesystem::directory_options::skip_permission_denied, ec);
169 !ec && it != std::filesystem::recursive_directory_iterator(); it.increment(ec)) {
170 if (!it->is_regular_file(ec) || it->path().extension() != missing.extension())
171 continue;
172 if (it->path().filename() == missing.filename())
173 sameName.push_back(it->path());
174 // Only files of the same size are hashed
175 if (fingerprint.isKnown() && it->file_size(ec) == fingerprint.size) {
176 if (const auto candidate = getFingerprint(toUtf8(it->path())); candidate && *candidate == fingerprint)
177 sameContent.push_back(it->path());
178 }
179 }
180
181 std::optional<std::filesystem::path> found;
182 bool byNameOnly = false;
183 if (sameContent.size() == 1) {
184 found = sameContent.front();
185 } else if (sameContent.size() > 1) {
186 // Identical copies: the one that kept its name, if only one did
187 std::vector<std::filesystem::path> named;
188 std::ranges::copy_if(sameContent, std::back_inserter(named), [&](const auto& path) { return path.filename() == missing.filename(); });
189 if (named.size() == 1)
190 found = named.front();
191 } else if (sameName.size() == 1) {
192 // No file with its content (it was edited as well as moved, or the scene was saved before fingerprints):
193 // the one file with its name, which the caller has to present as a guess
194 found = sameName.front();
195 byNameOnly = true;
196 }
197
198 if (!found)
199 return std::nullopt;
200 std::optional<std::string> newId = getProjectAssetId(*found);
201 if (!newId)
202 return std::nullopt;
203 return MovedAsset{std::move(*newId), byNameOnly};
204 }

206 void AssetsManager::forgetProjectAssets() {
207 if (m_projectAssets.empty())
208 return;
209 for (const auto& [type, id, cacheKey] : m_projectAssets) {
210 m_cache.erase(cacheKey);
211 switch (type) {
212 case TEXTURE: m_textureManager->remove(id); break;
213 case SHADER: m_shaderManager->remove(id); break;
214 case MESH: m_meshManager->remove(id); break;
215 default: break;
216 }
217 }
218 VC_CORE_INFO("Released the {} assets of the previous assets folder", m_projectAssets.size());
219 m_projectAssets.clear();
220 }
