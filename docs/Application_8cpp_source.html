22 // From the bottom up: the layers in the order they were pushed, then the overlays, so what is pushed later
23 // draws on top. By index, re-reading the size, and holding each layer during its call: a callback that
24 // pushes or pops a layer can't invalidate the loop (a layer pushed now is called this frame too)
25 template<typename Fn>
26 void forEachLayer(const LayerStack& stack, Fn&& fn) {
27 for (const std::vector<std::shared_ptr<Layer>>* group : {&stack.layers(), &stack.overlays()}) {
28 for (const auto& layer : *group) {
29 fn(layer);
30 }
31 }
32 }

38 VC_PROFILER_FUNCTION();
39 VC_CORE_ASSERT(!s_instance, "Application already exists!");
40 s_instance = this;

42 // Created and loaded before the window: the graphics backend reads settings
43 // during init(). A per-project load layers on top of this later (openScene).

107 void Application::dispatchEvent(Event& event) {
108 VC_PROFILER_FUNCTION();
109 // Who consumed the event, for the observers. A copy: the layer that did can be popped by its own handler
110 std::string consumedBy;
111
112 // The ImGuiLayer is owned apart from the stack but sits on top of everything: it gets the first look so
113 // it can swallow the mouse/keyboard events ImGui wants (see startBlockEvents)
114 if (m_imGuiLayer) {
115 if (m_imGuiLayer->capturesEvent(event))
116 event.m_handled = true;
117 m_imGuiLayer->notify(event);
118 if (event.m_handled)
119 consumedBy = "ImGui";
120 }
121
122 // From the top down: the last overlay pushed first, the first layer pushed last. By index, re-reading the
123 // size, and holding each layer during its call: a handler that pops a layer can't invalidate the loop
124 for (const std::vector<std::shared_ptr<Layer>>* group : {&m_layerStack.overlays(), &m_layerStack.layers()}) {
125 for (size_t i = group->size(); i-- > 0 && !event.m_handled;) {
126 if (i >= group->size())
127 continue; // a layer removed by an earlier handler
128 const std::shared_ptr<Layer> layer = (*group)[i];
129 layer->notify(event);
130 if (event.m_handled)
131 consumedBy = layer->getName();
132 }
133 }
134
135 if (!event.m_handled) {
136 notify(event);
137 if (event.m_handled)
138 consumedBy = "Application";
139 }
140
141 // Every listener sees a close request first: one that consumes it (e.g. to ask about unsaved work) keeps
142 // the application running
143 if (!event.m_handled && event.is<WindowCloseEvent>())
144 m_running = false;
145
146 detail::EventDispatchedEvent dispatched(event, consumedBy);
147 m_eventObservers.notify(dispatched);
148 }

150 void Application::run() {
151 VC_PROFILER_FUNCTION();
152 m_window->show();
153 Renderer::resizeMask({m_window->getWidth(),m_window->getHeight()});
154 while (m_running) {
155 const auto time = static_cast<float>(glfwGetTime());
156 m_deltaTime = time - m_LastFrameTime;
157 m_LastFrameTime = time;
158
159 if (m_hasToSwitch) {
160 // The previous frame can still be drawing with the old layer's resources, which go away with it
161 GraphicsContext::waitIdle();
162 // The new layer takes the old one's place, so the order of the other layers doesn't change
163 if (const std::shared_ptr<Layer> oldLayer = m_layerStack.ReplaceLayer(m_oldLayer, m_nextLayer))
164 oldLayer->OnDetach();
165 if (m_dataToNextLayer.empty()) m_nextLayer->OnAttach();
166 else m_nextLayer->OnAttach(m_dataToNextLayer);
167
168 m_oldLayer = nullptr;
169 m_nextLayer.reset();
170 m_dataToNextLayer = JsonObject();
171 m_hasToSwitch = false;
172 }
173
174 // What was posted since the last frame, the input polled at its end included
175 m_eventQueue.dispatch([this](Event& event) { dispatchEvent(event); });
176 if (!m_running)
177 break;
178
179 forEachLayer(m_layerStack, [this](const std::shared_ptr<Layer>& layer) { layer->OnUpdate(m_deltaTime); });
180 if (RenderCommand::canRender()) {
181 forEachLayer(m_layerStack, [](const std::shared_ptr<Layer>& layer) { layer->OnRenderOffscreen(); });
182
183 RenderCommand::beginFrame();
184 forEachLayer(m_layerStack, [](const std::shared_ptr<Layer>& layer) { layer->OnRender(); });
185 RenderCommand::endFrame();
186
187 m_imGuiLayer->OnRender();
188 m_imGuiLayer->OnUpdate(m_deltaTime);
189
190 RenderCommand::sendFrame();
191 }
192
193 m_window->onUpdate();
194 }
