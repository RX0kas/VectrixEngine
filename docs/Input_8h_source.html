10 /**

12 *
13 * Every function is static and answers about the current state, which makes it the
14 * counterpart of the event system: use an Event to react to the moment something
15 * changed, and Input to ask whether a key is held right now. The platform provides
16 * the implementation, so nothing has to be created by hand.
17 * @see KeyPressedEvent
18 * @see MouseEvent
19 * @ingroup input
20 */

23 virtual ~Input() = default;

29 * @see KeyCodes.h
30 */
31 static bool isKeyPressed(int keycode) { return s_instance->isKeyPressedImpl(keycode); }

33 /**

39 static bool isMouseButtonPressed(int button) { return s_instance->isMouseButtonPressedImpl(button); }
