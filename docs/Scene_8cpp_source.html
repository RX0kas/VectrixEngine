14 /// Reads the serialized fields of one component, refusing to read past its end (a truncated or
15 /// corrupt scene file must fail to load, not read out of bounds)
16 class ComponentReader {
17 public:
18 explicit ComponentReader(const std::vector<std::byte>& bytes) : m_bytes(bytes) {}
19
20 bool read(void* out, const size_t size) {
21 if (m_bytes.size() - m_offset < size) return false;
22 std::memcpy(out, m_bytes.data() + m_offset, size);
23 m_offset += size;
24 return true;
25 }
26
27 bool readFloat(float& out) { return read(&out, sizeof(float)); }
28
29 bool readBool(bool& out) {
30 std::uint8_t byte = 0; // read as a byte: a value other than 0/1 read straight into a bool is UB
31 if (!read(&byte, sizeof(byte))) return false;
32 out = byte != 0;
33 return true;
34 }
35
36 bool readString(std::string& out) {
37 std::uint16_t length = 0;
38 if (!read(&length, sizeof(length))) return false;
39 if (m_bytes.size() - m_offset < length) return false;
40 out.assign(reinterpret_cast<const char*>(m_bytes.data() + m_offset), length);
41 m_offset += length;
42 return true;
43 }
44
45 private:
46 const std::vector<std::byte>& m_bytes;
47 size_t m_offset = 0;
48 };

55 /// Loads the asset a MeshRendererComponent refers to. An empty id is an asset that was never set
56 /// (the component was saved half configured): it stays null. An asset that can't be loaded is reported
57 /// and its id kept in missingId, so saving writes the reference back: one moved or deleted file must
58 /// neither make the whole scene unopenable nor be dropped from it
59 template<typename T>
60 void loadComponentAsset(std::string id, std::shared_ptr<T>& out, std::string& missingId, const std::string& entityName,
61 const std::string& sceneName, std::vector<std::string>* assetErrors) {
62 if (id.empty()) return;
63 // Scenes saved on Windows before ids were normalised hold '\' separators, a plain character elsewhere
64 std::ranges::replace(id, '\\', '/');
65 auto [result, asset] = AssetsManager::load<T>(id);
66 if (result != SUCCESS) {
67 VC_CORE_WARN("Can't load {} for entity {} of scene {}: {}", id, entityName, sceneName, toString(result));
68 if (assetErrors)
69 assetErrors->push_back(std::format("{}: {} ({})", entityName, id, toString(result)));
70 missingId = std::move(id);
71 // A texture has a stand-in that shows something is missing; a mesh or shader doesn't
72 if constexpr (std::is_same_v<T, Texture>)
73 out = TextureManager::getNotFoundTexture();
74 return;
75 }
76 out = asset;
77 }
78 }

80 std::pair<VectrixResult, std::shared_ptr<Scene>> Scene::loadScene(SceneCreationData &creationData, std::vector<std::string>* assetErrors) {
81 if (creationData.result!=SUCCESS) {
82 return {creationData.result,nullptr};
83 }
84 std::shared_ptr<Scene> scene = std::make_shared<Scene>(creationData.name);
85
86 for (const auto& entityData : creationData.entities) {
87 auto entity = scene->createEntity(entityData.name);
88 for (const auto& componentData : entityData.components) {
89 ComponentReader reader(componentData.data);
90
91 if (entt::type_hash<TransformComponent>::value()==componentData.type_id) {
92 if (componentData.data.size() != sizeof(TransformComponent))
93 return malformed(creationData.name, entityData.name);
94 auto& t = entity->getComponent<TransformComponent>();
95 std::memcpy(&t, componentData.data.data(), sizeof(TransformComponent));
96 } else if (entt::type_hash<MeshRendererComponent>::value() == componentData.type_id) {
97 std::string meshId, textureId, shaderId;
98 bool enable = false;
99 if (!reader.readString(meshId) || !reader.readString(textureId) || !reader.readString(shaderId) || !reader.readBool(enable))
100 return malformed(creationData.name, entityData.name);
101
102 // Each asset's fingerprint follows in scenes saved since they exist (older ones end with the flag)
103 std::array<AssetFingerprint, 3> fingerprints{};
104 for (AssetFingerprint& fingerprint : fingerprints) {
105 if (!reader.read(&fingerprint.size, sizeof(fingerprint.size)) || !reader.read(&fingerprint.hash, sizeof(fingerprint.hash))) {
106 fingerprints = {};
107 break;
108 }
109 }
110
111 auto& m = entity->addComponent<MeshRendererComponent>();
112 loadComponentAsset(meshId, m.mesh, m.missingMesh, entityData.name, creationData.name, assetErrors);
113 loadComponentAsset(textureId, m.texture, m.missingTexture, entityData.name, creationData.name, assetErrors);
114 loadComponentAsset(shaderId, m.shader, m.missingShader, entityData.name, creationData.name, assetErrors);
115
116 // Kept for the missing ones: saved back, and what finds them again if they were moved
117 const std::array<const std::string*, 3> missing = {&m.missingMesh, &m.missingTexture, &m.missingShader};
118 for (size_t i = 0; i < missing.size(); ++i) {
119 if (!missing[i]->empty() && fingerprints[i].isKnown())
120 scene->m_missingFingerprints[*missing[i]] = fingerprints[i];
121 }
122
123 // Only fails when an asset it needs was left out above, which is already reported
124 if (enable && !m.tryEnabling()) {
125 m.enabledOnceComplete = true;
126 VC_CORE_WARN("Entity {} of scene {} left disabled: an asset it needs is missing", entityData.name, creationData.name);
127 }
128 } else if (entt::type_hash<CameraComponent>::value()==componentData.type_id) {
129 float fov = 0.0f, camNear = 0.0f, camFar = 0.0f, aspect = -1.0f;
130 if (!reader.readFloat(fov) || !reader.readFloat(camNear) || !reader.readFloat(camFar) || !reader.readFloat(aspect))
131 return malformed(creationData.name, entityData.name);
132
133 auto& c = entity->addComponent<CameraComponent>();
134 c.camera.setFOV(fov);
135 c.camera.setCamNear(camNear);
136 c.camera.setCamFar(camFar);
137 if (aspect > 0.0f) {
138 c.camera.setCustomAspect(aspect);
139 }
140 }
141 }
142 }
143 return {SUCCESS, scene};
144 }
