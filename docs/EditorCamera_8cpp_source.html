52 void EditorCamera::setViewDirection(glm::vec3 direction) {
53 VC_PROFILER_FUNCTION();
54 const glm::vec3 up = {0.0f, 1.0f, 0.0f};
55 const glm::vec3 w{glm::normalize(direction)};
56 const glm::vec3 u{glm::normalize(glm::cross(w, up))};
57 const glm::vec3 v{glm::cross(w, u)};
58
59 m_viewMatrix = glm::mat4{1.f};
60 m_viewMatrix[0][0] = u.x;
61 m_viewMatrix[1][0] = u.y;
62 m_viewMatrix[2][0] = u.z;
63 m_viewMatrix[0][1] = v.x;
64 m_viewMatrix[1][1] = v.y;
65 m_viewMatrix[2][1] = v.z;
66 m_viewMatrix[0][2] = w.x;
67 m_viewMatrix[1][2] = w.y;
68 m_viewMatrix[2][2] = w.z;
69 m_viewMatrix[3][0] = -glm::dot(u, m_position);
70 m_viewMatrix[3][1] = -glm::dot(v, m_position);
71 m_viewMatrix[3][2] = -glm::dot(w, m_position);
72
73 recalculateTransformationMatrix();
74 }
