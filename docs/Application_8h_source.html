106 /**
107 * @brief Post an event, sent to the listeners at the start of the next frame
108 *
109 * Works for the engine's events and for the application's own (see EventBase).
110 * @tparam T The event class
111 * @param args The arguments of T's constructor
112 * @note Can be called from any thread (e.g. an asset loading thread): the event is still sent on the
113 * main thread. The events of other threads come after the main thread's in the frame
114 * @see sendEvent
115 * @see EventListener::subscribe
116 */
117 template<std::derived_from<Event> T, typename... Args> requires std::constructible_from<T, Args...>
118 void postEvent(Args&&... args) { m_eventQueue.post<T>(std::forward<Args>(args)...); }
119
120 /**
121 * @brief Send an event to the listeners right away, before returning
122 *
123 * For when the answer is needed now, e.g. to ask whether a listener objects before doing something.
124 * The event goes through the same path as a posted one (ImGui, the layers from the top, then the
125 * application), so an unconsumed WindowCloseEvent closes the application.
126 * @tparam T The event class
127 * @param args The arguments of T's constructor
128 * @return Whether a listener consumed the event
129 * @note Main thread only. Prefer postEvent unless the answer matters: sending from inside a handler
130 * nests the dispatch
131 * @see postEvent
132 */
133 template<std::derived_from<Event> T, typename... Args> requires std::constructible_from<T, Args...>
134 bool sendEvent(Args&&... args) {
135 T event(std::forward<Args>(args)...);
136 dispatchEvent(event);
137 return event.isHandled();
138 }
139
140 /**
141 * @brief Call a function after each event was sent, whether a listener consumed it or not
142 *
143 * Meant for tools such as an event log: unlike a subscription, an observer sees every event, those
144 * ImGui or a layer consumed included, and is told who consumed it. It can't consume nor change it.
145 * @param observer Called with the event and the name of the listener that consumed it: "ImGui", a
146 * layer's name or "Application", empty when none did
147 * @return The handle keeping the observer: it stops being called when the handle is destroyed
148 * @note Main thread only, like the events. An observer may post events, which are only sent next frame
149 */
150 template<std::invocable<const Event&, std::string_view> F>
151 [[nodiscard]] ScopedSubscription observeEvents(F&& observer) {
152 return m_eventObservers.subscribeScoped<detail::EventDispatchedEvent>(
153 [o = std::forward<F>(observer)](const detail::EventDispatchedEvent& e) mutable { std::invoke(o, e.event, e.consumedBy); });
154 }
