18namespace Vectrix {
19 /**
20 * @brief A handle on one object of a Scene, and on the components it owns
21 *
22 * An entity is a lightweight handle rather than a container: it holds which scene it
23 * belongs to and an identifier, the components themselves live in the scene. Copying
24 * one is cheap and gives another handle on the same object. Build them with
25 * Scene::createEntity.
26 * @see Scene::createEntity
27 * @see InformationComponent
28 * @ingroup ecs
29 */
30 class Entity {
31 public:
32 /**
33 * @brief Build a handle pointing at nothing
34 * @see nullEntity
35 */
36 Entity() : m_scene(nullptr), m_entityHandle(entt::null){}
37
38 /**
39 * @brief This function add a component to the entity
40 * @pre The entity must not have the component
41 * @tparam T The component type
42 * @tparam Args The constructor type
43 * @param args Constructor arguments
44 * @return A reference to the component that was just added
45 */
46 template<typename T,typename... Args>
47 T& addComponent(Args&&... args) {
48 VC_CORE_ASSERT(!hasComponent<T>(), "Entity already has component");
49 return m_scene->m_registry.emplace<T>(m_entityHandle,m_scene->getEntity(m_entityHandle),std::forward<Args>(args)...);
50 }
51
52 /**
53 * @brief This function tell if the entity has the component
54 * @tparam T The component type
55 * @return true when the entity owns a component of that type
56 */
57 template<typename T>
58 [[nodiscard]] bool hasComponent() const {
59 return m_scene->m_registry.all_of<T>(m_entityHandle);
60 }
61
62 /**
63 * @brief This function remove a component from the entity
64 * @pre The entity must have the component
65 * @tparam T The component type
66 * @warning InformationComponent and TransformComponent cannot be removed, every
67 * entity is expected to have them
68 */
69 template<typename T>
70 void deleteComponent() {
71 VC_CORE_ASSERT(hasComponent<T>(), "Entity has not component");
72 bool isInfoComp = std::is_same_v<T, InformationComponent>;
73 bool isTransfComp = std::is_same_v<T, TransformComponent>;
74 VC_CORE_ASSERT(!isInfoComp,"Can't remove InformationComponent");
75 VC_CORE_ASSERT(!isTransfComp,"Can't remove TransformComponent");
76 m_scene->m_registry.erase<T>(m_entityHandle);
77 }
78
79 /**
80 * @brief This function return the component of the entity from a certain type
81 * @pre The entity must have the component
82 * @tparam T The component type
83 * @return A reference to the component, valid while the entity keeps it
84 * @see hasComponent
85 */
86 template<typename T>
87 T& getComponent() {
88 VC_CORE_ASSERT(hasComponent<T>(),"Entity has not component");
89 return m_scene->m_registry.get<T>(m_entityHandle);
90 }
91
92 /**
93 * @brief Checks if the entity is valid
94 * @return true when the handle points at an actual entity
95 */
96 operator bool() const { return m_entityHandle != entt::null; }
97
98 /**
99 * @brief Convert to the underlying registry handle
100 * @return The handle the entity wraps
101 */
102 operator entt::entity() const { return m_entityHandle; }
103
104 /**
105 * @brief Convert to a plain number, mostly to use as an identifier in the UI
106 * @return The handle as an unsigned integer
107 */
108 operator uint32_t() const { return static_cast<uint32_t>(m_entityHandle); }
109
110 /**
111 * @brief Tell if two handles point at the same entity of the same scene
112 * @param other The entity to compare with
113 * @return true when both describe the same object
114 */
115 bool operator==(const Entity& other) const
116 {
117 return m_entityHandle == other.m_entityHandle && m_scene == other.m_scene;
118 }
119
120 /**
121 * @brief Tell if two handles point at different entities
122 * @param other The entity to compare with
123 * @return true when they do not describe the same object
124 */
125 bool operator!=(const Entity& other) const
126 {
127 return !(*this == other);
128 }
129
130 /**
131 * @brief Return a handle pointing at nothing
132 * @return An entity that converts to false
133 */
134 static Entity nullEntity() { return {};}
135
136 /**
137 * @brief Return the identifier of the entity inside its scene
138 * @return The registry handle
139 */
140 [[nodiscard]] entt::entity getID() const { return m_entityHandle; }
141 private:
142 friend class Scene;
143 friend class SceneHierarchyPanel;
144 friend class EditorLayer;
145 friend class SceneSerializer;
146 Entity(entt::entity handle, Scene* scene);
147 Entity(const Entity& other) = default;
148 entt::entity m_entityHandle{ entt::null };
149 Scene* m_scene = nullptr;
150 };
